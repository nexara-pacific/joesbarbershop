```astro
---
// site/src/components/schema/HairSalon.astro
// Per D-06: auto-injected by Base.astro on every page.
// Per D-07: stable @id for cross-page entity dedup.
// Per D-04: includes geo, priceRange, openingHoursSpecification, sameAs.
import { business, toE164, toOpeningHoursSpecification, canonicalUrl } from '../../data/business';

const schema = {
  '@context': 'https://schema.org',
  '@type': 'HairSalon',
  '@id': `${canonicalUrl}/#business`,
  name: business.name,
  url: canonicalUrl,
  telephone: toE164(business.phone),
  priceRange: business.priceRange,
  image: `${canonicalUrl}/photos/03-interior-hero.jpg`,
  address: {
    '@type': 'PostalAddress',
    streetAddress: `${business.address.street} ${business.address.suite}`.trim(),
    addressLocality: business.address.city,
    addressRegion: business.address.state,
    postalCode: business.address.zip,
    addressCountry: 'US',
  },
  geo: {
    '@type': 'GeoCoordinates',
    latitude: business.geo.latitude,
    longitude: business.geo.longitude,
  },
  openingHoursSpecification: toOpeningHoursSpecification(business.hours),
  areaServed: business.areaServed.map((name) => ({ '@type': 'City', name })),
  sameAs: Object.values(business.sameAs).filter((u) => Boolean(u)),
};
---
<script type="application/ld+json" set:html={JSON.stringify(schema)} />
```

// site/src/components/schema/FAQPage.astro
// Per AEO-02: emit FAQPage wherever FAQs appear (homepage, niche-landing, cost guide,
// faq master, all 6 services, all 5 neighborhoods).
// Per RESEARCH Pitfall 5: emit only the page's own questions — no duplication.

interface FAQ {
  q: string;
  a: string;
}

interface Props {
  faqs: FAQ[];
  pageUrl: string;
}

const { faqs, pageUrl } = Astro.props;

const schema = {
  '@context': 'https://schema.org',
  '@type': 'FAQPage',
  '@id': `${pageUrl}#faq`,

    '@type': 'Question',
    name: q,
    acceptedAnswer: {
      '@type': 'Answer',
      text: a,
    },
  })),
};
---
<script type="application/ld+json" set:html={JSON.stringify(schema)} />

// site/src/components/schema/Service.astro
// Per AEO-03 + D-09: emitted by each service page via [service].astro.
// Per D-07: references HairSalon as provider via @id.
// Per CONTEXT Deferred: basic Offer (no full PriceSpecification — overkill for $30 ticket).
import { business, canonicalUrl } from '../../data/business';
import type { CollectionEntry } from 'astro:content';

interface Props {
  entry: CollectionEntry<'services'>;
  pageUrl: string;
}

const { entry, pageUrl } = Astro.props;

const schema = {
  '@context': 'https://schema.org',
  '@type': 'Service',
  '@id': `${pageUrl}#service`,
  name: entry.data.title,
  description: entry.data.bluf,
  provider: { '@id': `${canonicalUrl}/#business` },
  areaServed: business.areaServed.map((name) => ({ '@type': 'Place', name })),
  offers: {
    '@type': 'Offer',
    price: entry.data.price,
    priceCurrency: 'USD',
    availability: 'https://schema.org/InStock',
  },
};
---
<script type="application/ld+json" set:html={JSON.stringify(schema)} />

If `price: z.number()` (non-nullable), leave the code as-is.

// site/src/components/schema/Article.astro
// Per D-09: emitted by /east-county-traditional-barbershop + /2026-east-county-barbershop-cost-guide.
// Per D-07: publisher references HairSalon via @id.
// Per RESEARCH § specifics: include image to avoid Google Rich Results "Article needs image" warning.
import type { Article, WithContext } from 'schema-dts';
import { canonicalUrl } from '../../data/business';

interface Props {
  headline: string;
  description: string;
  url: string;
  datePublished: string;   // ISO 8601 (e.g., '2026-05-01')
  dateModified: string;    // ISO 8601 from git-mtimes.json (Plan 06)
  authorName: string;      // typically "Joe Denesowicz" or "Darrell Tang"
}

const { headline, description, url, datePublished, dateModified, authorName } = Astro.props;

const schema: WithContext<Article> = {
  '@context': 'https://schema.org',
  '@type': 'Article',
  headline,
  description,
  url,
  datePublished,
  dateModified,
  image: `${canonicalUrl}/photos/02-storefront.jpg`,
  author: { '@type': 'Person', name: authorName },
  publisher: {
    '@type': 'Organization',
    '@id': `${canonicalUrl}/#business`,
    name: "Joe's Barbershop",
  },
};
---
<script type="application/ld+json" set:html={JSON.stringify(schema)} />

// site/src/components/schema/Review.astro
// Per CONTEXT line 279: scoped IN for Phase 5. /reviews page only.
// Per D-07: each Review references the business via itemReviewed.@id.
import { canonicalUrl } from '../../data/business';

interface ReviewItem {
  name: string;
  rating: number;
  quote: string;
  source: string;   // "Google" or "Yelp"
  date: string;     // e.g., "Sep 2021" or ISO — Schema.org accepts year-month
}

interface Props {
  reviews: ReviewItem[];
}

const { reviews } = Astro.props;

const schemas = reviews.map((r) => ({
  '@context': 'https://schema.org',
  '@type': 'Review',
  itemReviewed: { '@id': `${canonicalUrl}/#business` },
  author: { '@type': 'Person', name: r.name },
  reviewRating: {
    '@type': 'Rating',
    ratingValue: r.rating,
    bestRating: 5,
    worstRating: 1,
  },
  reviewBody: r.quote,
  publisher: { '@type': 'Organization', name: r.source },
  datePublished: r.date,
}));
---
{schemas.map((s) => (
  <script type="application/ld+json" set:html={JSON.stringify(s)} />
))}
```

**Important Astro mapping syntax:** the iteration uses `schemas.map(...)` returning JSX elements. Each `<script>` element is rendered separately. This is valid Astro pattern (no `client:*` directive needed since rendering is purely static).
